Zone d'événements

Interagissez avec cette zone


// Gestionnaires d'événements modernes
const button = document.querySelector('.btn');

// Fonction de gestion
const handleClick = (event) => {
    event.preventDefault(); // Empêcher le comportement par défaut
    event.stopPropagation(); // Arrêter la propagation
    
    console.log('Bouton cliqué!', event.target);
};

// Ajout d'écouteurs
button.addEventListener('click', handleClick);
button.addEventListener('mouseover', (e) => {
    e.target.style.backgroundColor = '#ff6b6b';
});

// Suppression d'écouteurs
button.removeEventListener('click', handleClick);

Programmation asynchrone

Promises, async/await et gestion des API

Fetch API et async/await

Simulation d'appel API


// Fonction async moderne
const fetchUserData = async (userId) => {
    try {
        // Afficher un loader
        showLoader();
        
        // Appel API avec fetch
        const response = await fetch(`/api/users/${userId}`);
        
        // Vérifier la réponse
        if (!response.ok) {
            throw new Error(`Erreur HTTP: ${response.status}`);
        }
        
        // Parser les données JSON
        const userData = await response.json();
        
        // Traiter les données
        displayUserData(userData);
        
    } catch (error) {
        // Gestion d'erreur
        console.error('Erreur lors du fetch:', error);
        showError(error.message);
    } finally {
        // Masquer le loader
        hideLoader();
    }
};

// Requêtes parallèles
const fetchMultipleData = async () => {
    try {
        const [users, posts, comments] = await Promise.all([
            fetch('/api/users').then(r => r.json()),
            fetch('/api/posts').then(r => r.json()),
            fetch('/api/comments').then(r => r.json())
        ]);
        
        console.log({ users, posts, comments });
    } catch (error) {
        console.error('Erreur lors des requêtes parallèles:', error);
    }
};

Modules ES6+

Organisation du code avec import/export

Import et Export

Simulation de modules


// math.js - Module utilitaire
export const PI = 3.14159;

export function add(a, b) {
    return a + b;
}

export function multiply(a, b) {
    return a * b;
}

// Export par défaut
export default function calculator(operation, a, b) {
    switch (operation) {
        case 'add': return add(a, b);
        case 'multiply': return multiply(a, b);
        default: return 0;
    }
}

// app.js - Utilisation des modules
import calculator, { PI, add, multiply } from './math.js';

// Import avec alias
import { add as addition } from './math.js';

// Import de tout
import * as MathUtils from './math.js';

console.log(calculator('add', 5, 3)); // 8
console.log(PI); // 3.14159
console.log(MathUtils.multiply(4, 2)); // 8

Import dynamique

Chargement conditionnel


// Import dynamique avec async/await
const loadModule = async (moduleName) => {
    try {
        // Import conditionnel
        const module = await import(`./modules/${moduleName}.js`);
        
        // Utilisation du module chargé
        return module.default;
    } catch (error) {
        console.error('Erreur de chargement:', error);
    }
};

// Chargement conditionnel basé sur l'utilisateur
const initializeApp = async () => {
    const userPrefs = getUserPreferences();
    
    if (userPrefs.language === 'fr') {
        const frModule = await import('./i18n/fr.js');
        setLanguage(frModule.translations);
    }
    
    if (userPrefs.theme === 'dark') {
        const darkTheme = await import('./themes/dark.js');
        applyTheme(darkTheme.default);
    }
};

// Import avec destructuration
const { createUser, deleteUser } = await import('./api/users.js');
