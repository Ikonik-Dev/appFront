🚀 Introduction au JavaScript

JavaScript est un langage de programmation qui permet d'ajouter de l'interactivité aux pages web. Il peut modifier le contenu, réagir aux actions de l'utilisateur et communiquer avec des serveurs.

Comment ajouter JavaScript ?

1. JavaScript inline :

<button onclick="alert('Hello!')">Cliquez-moi</button>

2. JavaScript interne :

<script>
    console.log('Hello World!');
</script>

3. JavaScript externe :

<script src="script.js"></script>

⚡ Programmation asynchrone

Promises et async/await

Gérer les opérations asynchrones de manière élégante et lisible. 

// Promise basique
function chargerDonnees(url) {
    return new Promise((resolve, reject) => {
        fetch(url)
            .then(response => {
                if (!response.ok) {
                    throw new Error(`Erreur HTTP: ${response.status}`);
                }
                return response.json();
            })
            .then(data => resolve(data))
            .catch(error => reject(error));
    });
}

// Utilisation avec .then()
chargerDonnees('/api/users')
    .then(users => {
        console.log('Utilisateurs chargés:', users);
        return chargerDonnees('/api/posts');
    })
    .then(posts => {
        console.log('Articles chargés:', posts);
    })
    .catch(error => {
        console.error('Erreur:', error.message);
    });

// Version moderne avec async/await
async function chargerToutesLesDonnees() {
    try {
        const users = await chargerDonnees('/api/users');
        console.log('Utilisateurs:', users);
        
        const posts = await chargerDonnees('/api/posts');
        console.log('Articles:', posts);
        
        // Chargement en parallèle
        const [produits, commandes] = await Promise.all([
            chargerDonnees('/api/products'),
            chargerDonnees('/api/orders')
        ]);
        
        return { users, posts, produits, commandes };
    } catch (error) {
        console.error('Erreur lors du chargement:', error);
        throw error;
    }
}

Gestion d'erreurs avancée

// Retry automatique avec délai exponentiel
async function fetchAvecRetry(url, options = {}, maxRetries = 3) {
    for (let i = 0; i <= maxRetries; i++) {
        try {
            const response = await fetch(url, options);
            
            if (!response.ok) {
                throw new Error(`HTTP ${response.status}: ${response.statusText}`);
            }
            
            return await response.json();
        } catch (error) {
            if (i === maxRetries) {
                throw new Error(`Échec après ${maxRetries} tentatives: ${error.message}`);
            }
            
            // Délai exponentiel: 1s, 2s, 4s...
            const delai = Math.pow(2, i) * 1000;
            await new Promise(resolve => setTimeout(resolve, delai));
            console.log(`Tentative ${i + 2}/${maxRetries + 1} dans ${delai}ms...`);
        }
    }
}

// Timeout pour les promesses
function avecTimeout(promesse, timeout = 5000) {
    return Promise.race([
        promesse,
        new Promise((_, reject) => {
            setTimeout(() => reject(new Error('Timeout dépassé')), timeout);
        })
    ]);
}

        if (nouvelAge >= 0) {
            this.age = nouvelAge;
        } else {
            throw new Error("L'âge doit être positif");
        }
    }
    
    // Méthode statique
    static comparer(animal1, animal2) {

    }
    
    // Méthode privée
    #genererID() {
        return this.#id;
    }
}

    }

Les closures permettent aux fonctions d'accéder aux variables de leur environnement parent. 

// Factory function avec closure
function creerCompteur(valeurInitiale = 0) {
    let compteur = valeurInitiale;
    
    // La fonction retournée "ferme" sur la variable compteur
    return {
        incrementer: () => ++compteur,
        decrementer: () => --compteur,
        obtenir: () => compteur,
        reinitialiser: () => compteur = valeurInitiale
    };
}

Local Storage et Session Storage

// Gestion du localStorage avec gestion d'erreurs
class StorageManager {
    static set(key, value) {
        try {
            const data = {
                value: value,
                timestamp: Date.now(),
                expires: null
            };
            localStorage.setItem(key, JSON.stringify(data));
            return true;
        } catch (error) {
            console.error('Erreur localStorage:', error);
            return false;
        }
    }
    
    static get(key) {
        try {
            const item = localStorage.getItem(key);
            if (!item) return null;
            
            const data = JSON.parse(item);
            
            // Vérifier l'expiration
            if (data.expires && Date.now() > data.expires) {
                this.remove(key);
                return null;
            }
            
            return data.value;
        } catch (error) {
            console.error('Erreur lecture localStorage:', error);
            return null;
        }
    }
    
    static setWithExpiry(key, value, hoursToExpire) {
        try {
            const data = {
                value: value,
                timestamp: Date.now(),
                expires: Date.now() + (hoursToExpire * 60 * 60 * 1000)
            };
            localStorage.setItem(key, JSON.stringify(data));
            return true;
        } catch (error) {
            console.error('Erreur localStorage:', error);
            return false;
        }
    }
    
    static remove(key) {
        localStorage.removeItem(key);
    }
    
    static clear() {
        localStorage.clear();
    }
}

// Utilisation
StorageManager.set('user', { nom: 'Alice', email: 'alice@email.com' });
StorageManager.setWithExpiry('token', 'abc123', 24); // Expire dans 24h

const user = StorageManager.get('user');
console.log(user); // { nom: 'Alice', email: 'alice@email.com' }

Fetch API avancé et gestion des erreurs

// Client HTTP avec gestion complète
class HttpClient {
    constructor(baseURL = '', defaultHeaders = {}) {
        this.baseURL = baseURL;
        this.defaultHeaders = {
            'Content-Type': 'application/json',
            ...defaultHeaders
        };
    }
    
    async request(endpoint, options = {}) {
        const url = `${this.baseURL}${endpoint}`;
        const config = {
            headers: { ...this.defaultHeaders, ...options.headers },
            ...options
        };
        
        try {
            const response = await fetch(url, config);
            
            // Gestion des codes d'erreur HTTP
            if (!response.ok) {
                const errorData = await this.parseErrorResponse(response);
                throw new HttpError(response.status, errorData.message || response.statusText, errorData);
            }
            
            // Parse de la réponse selon le type de contenu
            const contentType = response.headers.get('content-type');
            if (contentType && contentType.includes('application/json')) {
                return await response.json();
            } else {
                return await response.text();
            }
            
        } catch (error) {
            if (error instanceof HttpError) {
                throw error;
            }
            throw new Error(`Erreur réseau: ${error.message}`);
        }
    }
    
    async parseErrorResponse(response) {
        try {
            return await response.json();
        } catch {
            return { message: response.statusText };
        }
    }
    
    get(endpoint, params = {}) {
        const url = new URL(endpoint, this.baseURL);
        Object.keys(params).forEach(key => url.searchParams.append(key, params[key]));
        return this.request(url.pathname + url.search);
    }
    
    post(endpoint, data) {
        return this.request(endpoint, {
            method: 'POST',
            body: JSON.stringify(data)
        });
    }
    
    put(endpoint, data) {
        return this.request(endpoint, {
            method: 'PUT',
            body: JSON.stringify(data)
        });
    }
    
    delete(endpoint) {
        return this.request(endpoint, { method: 'DELETE' });
    }
}

// Classe d'erreur personnalisée
class HttpError extends Error {
    constructor(status, message, data = null) {
        super(message);
        this.name = 'HttpError';
        this.status = status;
        this.data = data;
    }
}

// Utilisation
const api = new HttpClient('https://api.exemple.com');

async function chargerUtilisateurs() {
    try {
        const users = await api.get('/users', { page: 1, limit: 10 });
        console.log('Utilisateurs:', users);
        return users;
    } catch (error) {
        if (error instanceof HttpError) {
            console.error(`Erreur API ${error.status}:`, error.message);
        } else {
            console.error('Erreur:', error.message);
        }
    }
}
