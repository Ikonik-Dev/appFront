function loadDataXHR() {
    const xhr = new XMLHttpRequest();
    
    xhr.onreadystatechange = function() {
        if (xhr.readyState === 4) {
            if (xhr.status === 200) {
                const data = JSON.parse(xhr.responseText);
                displayResults(data);
            } else {
                console.error('Erreur:', xhr.status);
            }
        }
    };
    
    xhr.open('GET', 'https://jsonplaceholder.typicode.com/users/1');
    xhr.send();
}

// Fonction async moderne
async function loadUserProfile(userId) {
    try {
        // 1. Charger les informations utilisateur
        const userResponse = await fetch(`/api/users/${userId}`);
        
        if (!userResponse.ok) {
            throw new Error(`Utilisateur non trouvé: ${userResponse.status}`);
        }
        
        const user = await userResponse.json();
        
        // 2. Charger les posts en parallèle
        const [postsResponse, albumsResponse] = await Promise.all([
            fetch(`/api/users/${userId}/posts`),
            fetch(`/api/users/${userId}/albums`)
        ]);
        
        const posts = await postsResponse.json();
        const albums = await albumsResponse.json();
        
        // 3. Retourner le profil complet
        return {
            user,
            posts,
            albums,
            profileComplete: true
        };
        
    } catch (error) {
        console.error('Erreur lors du chargement:', error);
        throw error; // Re-lancer pour gestion en amont
    }
}

// Utilisation
async function displayProfile() {
    try {
        const profile = await loadUserProfile(1);
        renderProfilePage(profile);
    } catch (error) {
        showErrorMessage('Impossible de charger le profil');
    }
}

Gestion robuste des erreurs


async function robustFetch(url, options = {}) {
    const defaultOptions = {
        timeout: 10000, // 10 secondes
        retries: 3,
        retryDelay: 1000 // 1 seconde
    };
    
    const config = { ...defaultOptions, ...options };
    
    for (let attempt = 0; attempt <= config.retries; attempt++) {
        try {
            // Créer un AbortController pour le timeout
            const controller = new AbortController();
            const timeoutId = setTimeout(() => {
                controller.abort();
            }, config.timeout);
            
            const response = await fetch(url, {
                ...config,
                signal: controller.signal
            });
            
            clearTimeout(timeoutId);
            
            // Vérifier le statut de la réponse
            if (!response.ok) {
                const errorData = await response.text();
                throw new HTTPError(
                    `HTTP ${response.status}: ${response.statusText}`,
                    response.status,
                    errorData
                );
            }
            
            return response;
            
        } catch (error) {
            // Dernière tentative, on relance l'erreur
            if (attempt === config.retries) {
                throw formatError(error);
            }
            
            // Attendre avant la prochaine tentative
            await delay(config.retryDelay * (attempt + 1));
        }
    }
}

// Classes d'erreur personnalisées
class HTTPError extends Error {
    constructor(message, status, data) {
        super(message);
        this.name = 'HTTPError';
        this.status = status;
        this.data = data;
    }
}

function formatError(error) {
    if (error.name === 'AbortError') {
        return new Error('Requête interrompue (timeout)');
    }
    
    if (error instanceof TypeError) {
        return new Error('Erreur réseau - Vérifiez votre connexion');
    }
    
    return error;
}

CRUD complet avec une API REST


// Classe pour gérer une ressource REST
class UserAPI {
    constructor(baseUrl = 'https://api.example.com') {
        this.baseUrl = baseUrl;
        this.defaultHeaders = {
            'Content-Type': 'application/json'
        };
    }
    
    // CREATE - Créer un nouvel utilisateur
    async create(userData) {
        const response = await fetch(`${this.baseUrl}/users`, {
            method: 'POST',
            headers: this.defaultHeaders,
            body: JSON.stringify(userData)
        });
        
        return await response.json();
    }
    
    // READ - Lire tous les utilisateurs
    async getAll(params = {}) {
        const queryString = new URLSearchParams(params).toString();
        const url = `${this.baseUrl}/users${queryString ? '?' + queryString : ''}`;
        
        const response = await fetch(url);
        return await response.json();
    }
    
    // READ - Lire un utilisateur spécifique
    async getById(id) {
        const response = await fetch(`${this.baseUrl}/users/${id}`);
        
        if (!response.ok) {
            throw new Error(`Utilisateur ${id} non trouvé`);
        }
        
        return await response.json();
    }
    
    // UPDATE - Mettre à jour un utilisateur
    async update(id, userData) {
        const response = await fetch(`${this.baseUrl}/users/${id}`, {
            method: 'PUT',
            headers: this.defaultHeaders,
            body: JSON.stringify(userData)
        });
        
        return await response.json();
    }
    
    // DELETE - Supprimer un utilisateur
    async delete(id) {
        const response = await fetch(`${this.baseUrl}/users/${id}`, {
            method: 'DELETE'
        });
        
        return response.ok;
    }
}

// Utilisation
const userApi = new UserAPI();

// Créer un utilisateur
const newUser = await userApi.create({
    name: 'John Doe',
    email: 'john@example.com'
});

// Récupérer tous les utilisateurs avec pagination
const users = await userApi.getAll({ 
    page: 1, 
    limit: 10 
});
